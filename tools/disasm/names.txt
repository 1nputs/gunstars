# tools/gen_symbols.py in src/symbols_gen.h. A routine gets its name when it is ported.

# ---- work RAM
FFA204 v_stage4              # the stage times 4: 0 ruins, 4 mine, 8 battleship, $C dice palace, $10 base, $14 space, $18 final
FFA208 v_stagesCleared       # a bit for each of the four selectable stages
FFA400 v_player1             # the players' objects, $200 bytes each; vitality at +$24
FFA600 v_player2
FFC980 v_boss                # the boss object; vitality at +$24
FFEE00 v_hintCode            # the H-interrupt handler, copied here (the vector at $70 points at it)
FFF704 v_frameReady          # set when the frame's logic has finished; clear while it runs
FFF705 v_pauseFlags          # bit 7 paused, bit 6 pausing allowed, bits 0 / 1 pad 1 / pad 2 may pause
FFF70A v_padReleased         # pad 1's buttons released this frame (+1: pad 2's)
FFF706 v_padHeld             # pad 1's buttons held (v_padHeld + 1: pad 2's); bit 7 Start
FFF708 v_padPressed          # pad 1's buttons pressed this frame (+ 1: pad 2's)
FFF720 v_jobKind             # the running job's kind | $8000; 0: no job list running
FFF724 v_jobList
FFF74A v_hintEffect          # HintEffect's index (times 4)
FFF74E v_hintState
FFF750 v_hintLine            # the H-interrupt handlers' state: which split of the frame comes next
FFF754 v_jobBufferFull       # a job's 16 tiles wait in $FFB600 for the vertical blank's DMA
FFF7D0 v_vdpRegs             # the VDP's registers as written: a $8rvv word per register
FFF807 v_soundPause          # the sound driver's pause request: 1 pause, $80 resume
FFFF0C v_gameMode
FFFF06 v_padId               # the device ID each pad answered (PadId; +1: pad 2)
FFFF20 v_buttonMap           # the button A, B, C each read: pad 1 at +0, +2, +4, pad 2 at +1, +3, +5
FFE000 v_sprites             # the sprite table, DMAed to VRAM $F400 each vertical blank ($280 bytes)
FFE300 v_palette             # the palette as shown (64 colours), DMAed to CRAM
FFE380 v_paletteTarget       # the palette a fade starts from or arrives at
FFF700 v_dmaCommand          # a DMA's command, written out from here as two words
FFF70C v_dmaQueue            # the DMA queue's start: entries of 16 bytes below $F400, added downward
FFF710 v_hscrollSrc          # the horizontal scroll table's address (DMAed to VRAM $F000)
FFF714 v_vscrollSrc          # the vertical scroll table's address (DMAed to VSRAM)
FFF755 v_displayOn           # the palette is shown and H-interrupts run; clear: CRAM filled with v_fadeColor
FFF75C v_fadeSpeed           # added to v_fadeLevel each frame: below 0 fading in, above 0 fading out
FFF75E v_fadeLevel           # how far the palette is faded: the high byte is subtracted from (or added to) each component
FFFF08 v_randomSeed
FFFF28 v_fadeColor           # 0: fades go to black; else to white, and the colour CRAM is filled with while the display is off
FFB400 v_tilePixels          # one tile being decoded, a word per pixel, row by row ($80 bytes); also the RLE buffer ($400)
FFB600 v_jobBuffer           # up to 16 decoded tiles waiting for the vertical blank's DMA
FFF722 v_jobLength           # the current job's tiles, or bytes
FFF728 v_jobSrc
FFF72C v_jobDst              # a RAM address (sign-extended) or a VRAM address
FFF730 v_bitsLeft            # the tile decoder's bits left in v_bitBuffer; as a long, an RLE job's source end
FFF732 v_bitBuffer           # the tile decoder's bits, the next at the top
